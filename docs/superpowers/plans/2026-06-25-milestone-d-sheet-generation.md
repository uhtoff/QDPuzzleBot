"""Captured header layout for code-generated puzzle tabs (Milestone D).

Replaces the hidden "Tab template" / "Meta Tab template" tabs: the header cells
they carried (name/url, flavour, progress) are captured once from the live
templates (see scripts/capture_template_layout.py) and stored here as data, so
every puzzle tab is generated in code.

A cell spec is {"row": int, "col": int, "value": <str|None>, "type": "stringValue"
| "formulaValue"}. `value` None means "leave blank / bot fills at write time"
(the name/url and answer cells are written dynamically, not from this artifact).
"""

# Filled by scripts/capture_template_layout.py. Until the capture runs these are
# empty and the generator writes only the dynamic cells (name/url, answer).
PUZZLE_TAB_CELLS: list[dict] = []
META_TAB_EXTRA_CELLS: list[dict] = []   # only the columns a target tab adds


def puzzle_tab_layout(is_target: bool) -> list[dict]:
    """Return the static header cell specs for a puzzle tab.

    is_target adds the meta-only header cells; the contribution Table itself is
    added separately by the generator, not from this artifact.
    """
    cells = list(PUZZLE_TAB_CELLS)
    if is_target:
        cells += META_TAB_EXTRA_CELLS
    return cells
```

"""One-off: dump the live template tabs' header grids into bot/cogs/sheet_layout.py.

Run once with live Google credentials + the hunt spreadsheet id holding the
templates. Reads each template tab's used range and emits cell specs for the
non-empty static cells (skipping the dynamic name/url + answer cells, which the
generator writes at runtime).

Usage:
    pipenv run python scripts/capture_template_layout.py <spreadsheet_id>
"""
import json
import sys

from googleapiclient.discovery import build  # noqa: F401  (uses existing creds path)
from bot.cogs.puzzles_gsheet import get_sheet  # the project's authed Sheets handle


TEMPLATES = {"Tab template": "PUZZLE_TAB_CELLS",
             "Meta Tab template": "META_TAB_EXTRA_CELLS"}


def _page_id(meta, title):
    for s in meta["sheets"]:
        if s["properties"]["title"] == title:
            return s["properties"]["sheetId"]
    raise SystemExit(f"template tab not found: {title}")


def main(spreadsheet_id):
    svc = get_sheet()
    meta = svc.get(spreadsheetId=spreadsheet_id,
                   includeGridData=True,
                   ranges=list(TEMPLATES)).execute()
    out = {}
    for sheet in meta["sheets"]:
        title = sheet["properties"]["title"]
        if title not in TEMPLATES:
            continue
        cells = []
        for r, row in enumerate(sheet.get("data", [{}])[0].get("rowData", [])):
            for c, cell in enumerate(row.get("values", [])):
                uev = cell.get("userEnteredValue")
                if not uev:
                    continue
                if "formulaValue" in uev:
                    cells.append({"row": r, "col": c,
                                  "value": uev["formulaValue"], "type": "formulaValue"})
                elif "stringValue" in uev:
                    cells.append({"row": r, "col": c,
                                  "value": uev["stringValue"], "type": "stringValue"})
        out[TEMPLATES[title]] = cells
    print(json.dumps(out, indent=2))
    print("\n# Paste the arrays above into bot/cogs/sheet_layout.py", file=sys.stderr)


if __name__ == "__main__":
    main(sys.argv[1])
```

```python
"""Pure helpers for the unified contribution table (Milestone D).

One table per target group. Each row is [key, type, source_name, content]:
  feeds:<pid>            Feeder       member name   solved answer (blank if unsolved)
  none:<pid>             Group        member name   ""
  informs:<pid>          Information  member name   ""   (bare edge, nothing logged)
  informs:<pid>:<eid>    Information  member name   datum content (one per evidence row)

Keys are edge-anchored and stable, so the column-A diff survives re-tagging.
"""


def _feeder_answer(puzzle):
    return puzzle.solution.upper() if (puzzle.solution) else ""


def contribution_rows(tag_group_id, members, evidence):
    """Build the desired contribution rows for one target group.

    members  : PuzzleData list (everything tagged into tag_group_id), each with
               .tag_edges[tag_group_id] in {"feeds","informs","none"}.
    evidence : list of dicts {"id","source_puzzle_id","content"} routed to this
               target (may be empty). Used only for informs members.
    Order: members in the given order; informs evidence in the given order.
    """
    ev_by_source = {}
    for e in evidence:
        ev_by_source.setdefault(e["source_puzzle_id"], []).append(e)

    rows = []
    for p in members:
        edge = p.tag_edges.get(tag_group_id)
        if edge == "feeds":
            rows.append([f"feeds:{p.id}", "Feeder", p.name, _feeder_answer(p)])
        elif edge == "none":
            rows.append([f"none:{p.id}", "Group", p.name, ""])
        elif edge == "informs":
            data = ev_by_source.get(p.id, [])
            if not data:
                rows.append([f"informs:{p.id}", "Information", p.name, ""])
            else:
                for e in data:
                    rows.append([f"informs:{p.id}:{e['id']}", "Information",
                                 p.name, e["content"]])
    return rows
```

Add to `GoogleSheets`, factoring the body-mutation logic out of `apply_unassigned_changes`.
`desired_rows` are `[key, *data...]`; the diff is computed with `diff_rows`. Writes the full
row (key in col A + data) so the column-A keys stay present for the next read.

```python
    async def reconcile_table(self, page_id, key_count, desired_rows, data_col_count):
        """Incrementally reconcile a key-anchored table on `page_id`.

        Reads column A keys, diffs against desired_rows (via diff_rows), then
        applies updates (rewrite cols A..data_col_count-1 on the matched row),
        prunes (delete whole row, bottom-first), and appends (write new rows
        below). key_count is unused beyond documenting the single key column;
        kept for symmetry with multi-key callers. Inserts only — never a sized
        overwrite of solver rows below the table."""
        from bot.core.unassigned import diff_rows
        spreadsheet_id = self.get_spreadsheet_id()

        tab_name = None
        for s in self.sheet_list(spreadsheet_id)["sheets"]:
            if s["properties"]["sheetId"] == int(page_id):
                tab_name = s["properties"]["title"]
                break
        if tab_name is None:
            return
        safe_name = tab_name.replace("'", "''")

        resp = await self._values_get(spreadsheet_id, f"'{safe_name}'!A:A")
        vals = resp.get("values", []) or []
        key_to_row = {}
        for i in range(1, len(vals)):
            if vals[i]:
                key_to_row[vals[i][0]] = i

        current_keys = list(key_to_row)
        appends, updates, prunes = diff_rows(current_keys, desired_rows)

        def _cell(v):
            return {"userEnteredValue": {"stringValue": str(v)}}

        requests = []
        # Updates: rewrite the whole row (cols 0..data_col_count-1).
        update_by_key = {row[0]: row for row in updates}
        for key, grid_row in key_to_row.items():
            row = update_by_key.get(key)
            if row is None:
                continue
            requests.append({"updateCells": {
                "range": {"sheetId": page_id, "startRowIndex": grid_row,
                          "endRowIndex": grid_row + 1, "startColumnIndex": 0,
                          "endColumnIndex": data_col_count},
                "fields": "userEnteredValue",
                "rows": [{"values": [_cell(v) for v in row]}],
            }})
        # Prunes: delete whole rows bottom-first.
        for grid_row in sorted((key_to_row[k] for k in prunes), reverse=True):
            requests.append({"deleteDimension": {
                "range": {"sheetId": page_id, "dimension": "ROWS",
                          "startIndex": grid_row, "endIndex": grid_row + 1}}})
        # Appends: write new rows below the current data (after prunes shrink it).
        if appends:
            start = len(vals) - len(prunes)
            requests.append({"updateCells": {
                "range": {"sheetId": page_id, "startRowIndex": start,
                          "endRowIndex": start + len(appends), "startColumnIndex": 0,
                          "endColumnIndex": data_col_count},
                "fields": "userEnteredValue",
                "rows": [{"values": [_cell(v) for v in row]} for row in appends],
            }})

        if requests:
            await self.batch_update({"requests": requests})
```

    async def promote_tab_to_target(self, puzzle):
        """Transform an existing puzzle tab into a target tab without losing
        solver data. Insert contribution-width columns at the left (shifting the
